function toggleAgentPanel() { var panel = document.getElementById("agent-panel"); if (panel) panel.classList.toggle("hidden"); } function toggleTheme() { var root = document.documentElement; var next = root.getAttribute("data-theme") === "dark" ? "light" : "dark"; try { localStorage.setItem("gi-theme", next); } catch (e) {} // Recargar para que los gráficos se redibujen con la paleta del tema. location.reload(); } var _agentSizes = ["small", "medium", "large"]; function cycleAgentPanelSize() { var panel = document.getElementById("agent-panel"); if (!panel) return; var idx = _agentSizes.indexOf(panel.dataset.size || "medium"); idx = (idx + 1) % _agentSizes.length; panel.dataset.size = _agentSizes[idx]; panel.classList.remove("small", "medium", "large"); panel.classList.add(_agentSizes[idx]); } // ── Chat optimista ───────────────────────────────────────────────── // El mensaje del usuario aparece al instante + indicador "escribiendo…". function appendAgentBubble(role, text) { var thread = document.getElementById("agent-chat-thread"); if (!thread) return; var d = document.createElement("div"); d.className = "msg msg-" + role; d.textContent = text; thread.appendChild(d); thread.scrollTop = thread.scrollHeight; } function appendAgentTyping() { var thread = document.getElementById("agent-chat-thread"); if (!thread || document.getElementById("agent-typing")) return; var d = document.createElement("div"); d.id = "agent-typing"; d.className = "msg msg-assistant agent-typing"; d.innerHTML = '<span class="dot"></span><span class="dot"></span><span class="dot"></span>'; thread.appendChild(d); thread.scrollTop = thread.scrollHeight; } // Enviar el chat: el form usa hx-post (htmx garantiza el envío). Este // listener hace el append optimista + indicador, y limpia el input en un // microtask para no interferir con la lectura del form por htmx. var _agentForm = document.getElementById("agent-chat-form"); if (_agentForm) { _agentForm.addEventListener("submit", function () { var input = _agentForm.querySelector('textarea[name=message]'); var msg = (input && input.value || "").trim(); if (!msg) return; appendAgentBubble("user", msg); // el mensaje va al historial YA appendAgentTyping(); // indicador "escribiendo…" setTimeout(function () { if (input) input.value = ""; }, 0); }, true); // capture: corre antes de que htmx lea el formulario // Enter envía; Shift+Enter inserta salto de línea (textarea multilínea). var _agentInput = _agentForm.querySelector('textarea[name=message]'); if (_agentInput) { _agentInput.addEventListener("keydown", function (evt) { if (evt.key === "Enter" && !evt.shiftKey) { evt.preventDefault(); _agentForm.requestSubmit(); } }); } } // Al terminar la respuesta: quitar el indicador; si fue un error (4xx), // htmx no swapea >=400, así que insertamos el reply de error a mano (B1). document.body.addEventListener("htmx:afterRequest", function (evt) { var cfg = evt.detail.requestConfig || {}; if (cfg.path !== "/ui/agent/chat") return; var typing = document.getElementById("agent-typing"); if (typing) typing.remove(); var xhr = evt.detail.xhr; if (xhr && xhr.status >= 400 && xhr.responseText) { var thread = document.getElementById("agent-chat-thread"); if (thread) thread.insertAdjacentHTML("beforeend", xhr.responseText); } }); // Mostrar la burbuja SOLO en el Dashboard Dinámico. Solo se actualiza al // cambiar de pestaña (swap en #tab-content); la respuesta del chat (swap // en #agent-chat-thread) NO debe ocultar la burbuja. document.body.addEventListener("htmx:afterSwap", function (evt) { var target = evt.detail.target; if (target && target.id === "tab-content") { var path = (evt.detail.requestConfig && evt.detail.requestConfig.path) || ""; var onDashboard = (path === "/ui/tabs/agent"); var fab = document.getElementById("agent-fab"); if (fab) fab.style.display = onDashboard ? "flex" : "none"; if (!onDashboard) { var panel = document.getElementById("agent-panel"); if (panel) panel.classList.add("hidden"); } } if (target && target.classList && target.classList.contains("agent-thread")) { target.scrollTop = target.scrollHeight; } });
</script>
